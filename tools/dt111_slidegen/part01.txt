const FONT='Noto Sans';

function deckBase(title){
  const pptx = new pptxgen();
  pptx.layout='LAYOUT_WIDE';
  pptx.author='Edwin John Kambo';
  pptx.subject='DT 111 Introduction to Data Science and AI';
  pptx.title=title;
  pptx.company='Mkwawa University College of Education';
  pptx.lang='en-GB';
  pptx.theme={headFontFace:FONT,bodyFontFace:FONT,lang:'en-GB'};
  pptx.defineSlideMaster({
    title:'CONTENT',
    background:{color:C.white},
    objects:[
      {rect:{x:0,y:0,w:W,h:0.12,fill:{color:C.blue},line:{color:C.blue}}},
      {text:{text:'DT 111 · Introduction to Data Science and AI',options:{x:0.55,y:7.16,w:6.4,h:0.18,fontFace:FONT,fontSize:9,color:'6B778C',margin:0}}},
      {text:{text:'MUCE · University of Dar es Salaam',options:{x:8.0,y:7.16,w:4.75,h:0.18,fontFace:FONT,fontSize:9,color:'6B778C',align:'right',margin:0}}}
    ],
    slideNumber:{x:12.72,y:7.14,w:0.25,h:0.18,color:'6B778C',fontFace:FONT,fontSize:9,align:'right'}
  });
  return pptx;
}
function addText(slide,text,x,y,w,h,size=24,color=C.ink,opts={}){
  slide.addText(text,{x,y,w,h,fontFace:FONT,fontSize:size,color,bold:!!opts.bold,align:opts.align||'left',valign:opts.valign||'top',margin:opts.margin??0,breakLine:false,fit:'shrink',bullet:opts.bullet,italic:!!opts.italic,paraSpaceAfterPt:opts.paraSpaceAfterPt||0});
}
function addTitle(slide,title,subtitle){
  addText(slide,title,0.62,0.42,12.0,0.68,30,C.navy,{bold:true});
  if(subtitle) addText(slide,subtitle,0.64,1.14,11.75,0.46,15,C.gray,{});
}
function addKicker(slide,text,x=0.65,y=0.32,w=4.0){
  slide.addShape('roundRect',{x,y,w,h:0.33,rectRadius:0.08,fill:{color:C.paleBlue},line:{color:C.paleBlue}});
  addText(slide,text.toUpperCase(),x+0.12,y+0.07,w-0.24,0.16,10,C.blue,{bold:true});
}

function cardsSlide(pptx,title,cards,subtitle){
  const s=pptx.addSlide('CONTENT'); addTitle(s,title,subtitle);
  const n=cards.length; const cols=n<=3?n:(n===4?2:3); const rows=Math.ceil(n/cols);
  const gap=0.24; const x0=0.72; const y0=1.78; const totalW=11.9; const totalH=4.95;
  const cw=(totalW-gap*(cols-1))/cols; const ch=(totalH-gap*(rows-1))/rows;
  cards.forEach((c,i)=>{
    const col=i%cols,row=Math.floor(i/cols); const x=x0+col*(cw+gap), y=y0+row*(ch+gap);
    const fill=c.fill||[C.paleBlue,C.paleTeal,C.paleOrange,C.paleGreen,C.paleRed][i%5];
    s.addShape('roundRect',{x,y,w:cw,h:ch,rectRadius:0.12,fill:{color:fill},line:{color:C.line,width:1}});
    if(c.tag){ addText(s,c.tag,x+0.22,y+0.18,cw-0.44,0.22,10,c.color||C.blue,{bold:true}); }
    addText(s,c.title,x+0.22,y+(c.tag?0.52:0.3),cw-0.44,0.46,19,C.navy,{bold:true});
    addText(s,c.body,x+0.22,y+(c.tag?1.02:0.85),cw-0.44,ch-(c.tag?1.2:1.05),14.5,C.ink,{});
  });
  return s;
}
function compareSlide(pptx,title,left,right,subtitle){

  boxes.forEach(({x,c,fill})=>{

    addText(s,c.title,x+0.28,2.15,5.25,0.48,23,C.navy,{bold:true});

}
function processSlide(pptx,title,steps,subtitle){
  const s=pptx.addSlide('CONTENT'); addTitle(s,title,subtitle);
  const n=steps.length; const gap=0.18; const x0=0.72; const y=2.3; const total=11.85; const w=(total-gap*(n-1))/n;
  steps.forEach((st,i)=>{
    const x=x0+i*(w+gap); const fill=[C.paleBlue,C.paleTeal,C.paleOrange,C.paleGreen,C.paleRed][i%5];
    s.addShape('roundRect',{x,y,w,h:2.4,rectRadius:0.12,fill:{color:fill},line:{color:C.line}});
    s.addShape('ellipse',{x:x+w/2-0.28,y:y-0.35,w:0.56,h:0.56,fill:{color:C.navy},line:{color:C.navy}});
    addText(s,String(i+1),x+w/2-0.14,y-0.22,0.28,0.24,12,C.white,{bold:true,align:'center'});
    addText(s,st.title,x+0.16,y+0.46,w-0.32,0.42,16,C.navy,{bold:true,align:'center'});
    addText(s,st.body,x+0.18,y+1.05,w-0.36,1.0,13.5,C.ink,{align:'center'});
    if(i<n-1) addText(s,'→',x+w+0.02,y+0.88,gap+0.14,0.4,18,C.gray,{bold:true,align:'center'});
  });
  return s;
}
