function addText(slide,text,x,y,w,h,size=24,color=C.ink,opts={}){

}

function cardsSlide(pptx,title,cards,subtitle){
  const s=pptx.addSlide('CONTENT'); addTitle(s,title,subtitle);
  const n=cards.length; const cols=n<=3?n:(n===4?2:3); const rows=Math.ceil(n/cols);
  const gap=0.24; const x0=0.72; const y0=1.78; const totalW=11.9; const totalH=4.95;
  const cw=(totalW-gap*(cols-1))/cols; const ch=(totalH-gap*(rows-1))/rows;
  cards.forEach((c,i)=>{
    const col=i%cols,row=Math.floor(i/cols); const x=x0+col*(cw+gap), y=y0+row*(ch+gap);
    const fill=c.fill||[C.paleBlue,C.paleTeal,C.paleOrange,C.paleGreen,C.paleRed][i%5];
    s.addShape('roundRect',{x,y,w:cw,h:ch,rectRadius:0.12,fill:{color:fill},line:{color:C.line,width:1}});
    if(c.tag){ addText(s,c.tag,x+0.22,y+0.18,cw-0.44,0.22,10,c.color||C.blue,{bold:true}); }
    addText(s,c.title,x+0.22,y+(c.tag?0.52:0.3),cw-0.44,0.46,19,C.navy,{bold:true});
    addText(s,c.body,x+0.22,y+(c.tag?1.02:0.85),cw-0.44,ch-(c.tag?1.2:1.05),14.5,C.ink,{});
  });
  return s;
}

function processSlide(pptx,title,steps,subtitle){
  const s=pptx.addSlide('CONTENT'); addTitle(s,title,subtitle);
  const n=steps.length; const gap=0.18; const x0=0.72; const y=2.3; const total=11.85; const w=(total-gap*(n-1))/n;
  steps.forEach((st,i)=>{
    const x=x0+i*(w+gap); const fill=[C.paleBlue,C.paleTeal,C.paleOrange,C.paleGreen,C.paleRed][i%5];
    s.addShape('roundRect',{x,y,w,h:2.4,rectRadius:0.12,fill:{color:fill},line:{color:C.line}});
    s.addShape('ellipse',{x:x+w/2-0.28,y:y-0.35,w:0.56,h:0.56,fill:{color:C.navy},line:{color:C.navy}});
    addText(s,String(i+1),x+w/2-0.14,y-0.22,0.28,0.24,12,C.white,{bold:true,align:'center'});
    addText(s,st.title,x+0.16,y+0.46,w-0.32,0.42,16,C.navy,{bold:true,align:'center'});
    addText(s,st.body,x+0.18,y+1.05,w-0.36,1.0,13.5,C.ink,{align:'center'});
    if(i<n-1) addText(s,'→',x+w+0.02,y+0.88,gap+0.14,0.4,18,C.gray,{bold:true,align:'center'});
  });
  return s;
}
